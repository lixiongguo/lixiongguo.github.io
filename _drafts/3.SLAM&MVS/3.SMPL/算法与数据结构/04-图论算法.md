	SElemType se;

typedef struct stack

  PNode top;

void push(Stack *ps,SElemType se)
{
  PNode pnode=(Node *)malloc(sizeof(Node));
  if(pnode!=NULL)
  {
	 pnode->se=se;
	 pnode->down=ps->top;
	 ps->size++;
	 ps->top=pnode;

  }
}
SElemType pop(Stack *ps)
{
  PNode p=ps->top;
  SElemType se=p->se;
  ps->size--;
  ps->top=p->down;
  free(p);
  return se;
}

{
   Stack *ps=(Stack *)malloc(sizeof(Stack));
   if(ps!=NULL){ps->top=NULL;ps->size=0;}

}

{
	if((ps->top==NULL)&&(ps->size==0))
	return 1;
	else return 0;
}

{
	pos[0][0].x=0;       pos[0][1].x=0;     	pos[0][2].x=0;       pos[0][3].x=0;

posType NextPos(Maze pos ,posType opos,int di)
{
	posType npos=opos;
	int x=opos.x;
	int y=opos.y;
	switch(di)
	{
    case 1: npos=pos[x][++y];break;
    case 2: npos=pos[++x][y];break;
    case 3: npos=pos[x][--y];break;
    case 4: npos=pos[--x][y];break;
	}

   posType start,end;
   posType curpos;

   SElemType e;

   end.x=pos[3][3].x;end.y=pos[3][3].y;end.pass=pos[3][3].pass;end.footprint=pos[3][3].footprint;

   curpos=start;

   do
   {
	   if(curpos.pass==1)
	   {
        curpos.footprint=1;
        pos[curpos.x][curpos.y].footprint=1;
	    e.pos=curpos;
     	e.order=curstep;
	    e.di=1;
		push(ps,e);
		if((curpos.x==end.x)&&(curpos.y==end.y))return 1;
	    curpos=NextPos(pos,curpos,1);
		curstep++;
	   }
	   else
	   {
          if(!IsEmpty(ps)) 
		  {
			  e=pop(ps);
		   while((e.di==4)&&(!IsEmpty(ps)))
		   {
                e.pos.pass=-1;
               pos[e.pos.x][e.pos.y].pass=-1;
				e=pop(ps);
		   }
		   if(e.di<4)
		   {
			    e.di++; push(ps,e);
			   curpos=NextPos(pos,e.pos,e.di);
		   }
			  
          }
	   }
   }while(!IsEmpty(ps));
   return 0;
}

main()
{

  Maze pos;
  posType p;
  MazeConstruct(pos);
if(RunMaze(pos,ps))
PrintMaze(pos);

}
