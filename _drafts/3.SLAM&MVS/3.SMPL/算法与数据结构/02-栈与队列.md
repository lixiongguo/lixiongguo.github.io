```c
#include<stdio.h>
typedef struct QNode
{
	int data;
	struct QNode *next;
}QNode,*QueuePtr;
typedef struct
{
  QueuePtr front;
  QueuePtr rear;
}LinkQueue;
LinkQueue *InitQueue()
{
  LinkQueue *q;
  q=(LinkQueue *)malloc(sizeof(LinkQueue));
  q->front=q->rear=NULL;
  return q;
}
void enQueue(LinkQueue *q,int e)
{
	QNode *s;
	s=(QNode *)malloc(sizeof(QNode));
	s->data=e;
	s->next=NULL;
	if(q->rear==NULL)
	{
		q->front=s;
		q->rear=s;
		s->next=NULL;
	}
	else
	{
		q->rear->next=s;
		q->rear=s;
		s->next=NULL;
	}
	
}
int deQueue(LinkQueue *q)
{
	int i;
	QNode *t;
	if(q->rear==NULL)printf("the queue is empty");
	if(q->front==q->rear)//只有一个元素时
	{
		t=q->front;
		q->front=NULL;
		q->rear=NULL;
	}
	else
	{
		t=q->front;
		q->front=q->front->next;
	}
	i=t->data;
	free(t);
	return i;
}
int QueEmpty(LinkQueue *q)
{
	if(q->rear==NULL)return 1;
	else return 0;
}
int QueLength(LinkQueue *q)
{
   int i=0;
   QueuePtr p=q->front;
   while(p!=NULL)
   {
	   p=p->next;
	   i++;
   }
   return i;
}
main()
{
  LinkQueue *lq;
  lq=InitQueue();
  enQueue(lq,1);enQueue(lq,2);
  printf("The Length of the queue is %d\n",QueLength(lq));
 while(!QueEmpty(lq))printf(" %d ",deQueue(lq));
}
```

## 利用栈实现简单数制转换

`源文件：数据结构实验课作业\利用栈实现简单数制转换\test.c`

```c
#include<stdio.h>
typedef struct node
{
	int data;
	struct node *down;
}Node,*PNode;
typedef struct stack
{
  PNode top;
  int size;
}Stack;
void Push(Stack *ps,int data)
{
	PNode pnode=(Node *)malloc(sizeof(Node));
	if(pnode!=NULL)
	{
		pnode->data=data;
		pnode->down=ps->top;//压入栈顶
		ps->size++;
		ps->top=pnode;
	}
}
int Pop(Stack *ps)
{
	PNode p=ps->top;
	int t=p->data;
	ps->size--;
	ps->top=ps->top->down;//栈顶弹出
	free(p);
	return t;
}
int IsEmpty(Stack *ps)
{
	if((ps->top==NULL)&&(ps->size==0))
	return 1;
	else return 0;
}
Stack *InitStack()
{
	Stack *ps=(Stack *)malloc(sizeof(Stack));
	if(ps!=NULL){ps->top=NULL;ps->size=0;}
    return ps;
}
main()
{
	int n;
    Stack *ps=InitStack();
printf("请输入一个十进制数：");
scanf("%d",&n);
	
while(n/2!=0)
	{
		Push(ps,n%2);
        n=n/2;
	}
Push(ps,n);
	printf("该数对应的二进制数是：");
   while(!IsEmpty(ps))
	{
		printf("%d",Pop(ps));
	}
   printf("\n");
}
```

```c
  LinkQueue *q;
  q=(LinkQueue *)malloc(sizeof(LinkQueue));
  q->front=q->rear=NULL;

}

{
	if(q->rear==NULL)
	{
		q->front=s;
		q->rear=s;
		s->next=NULL;
	}
	else
	{
		q->rear->next=s;
		q->rear=s;
		s->next=NULL;

}

{
	QNode *t;

		q->front=NULL;
		q->rear=NULL;

	{
		t=q->front;
		q->front=q->front->next;
	}
	return t;
}

	if(q->rear==NULL)return 1;

   int i=0;
```

```c
  LinkQueue *q;

  return q;
}
void enQueue(LinkQueue *q,int e)
{
	QNode *s;
	s=(QNode *)malloc(sizeof(QNode));
	s->data=e;
	s->next=NULL;
	if(q->rear==NULL)
	{
		q->front=s;
		q->rear=s;
		s->next=NULL;
	}
	else
	{
		q->rear->next=s;
		q->rear=s;
		s->next=NULL;
	}
	
}
int deQueue(LinkQueue *q)
{
	int i;
	QNode *t;
	if(q->rear==NULL)printf("the queue is empty");
	if(q->front==q->rear)//只有一个元素时
	{
		t=q->front;
		q->front=NULL;
		q->rear=NULL;
	}
	else
	{
		t=q->front;
		q->front=q->front->next;
	}
	i=t->data;
	free(t);
	return i;
}

	if(q->rear==NULL)return 1;
	else return 0;
}

   int i=0;
   QueuePtr p=q->front;

   {
	   p=p->next;
	   i++;
   }
   return i;
}
```
