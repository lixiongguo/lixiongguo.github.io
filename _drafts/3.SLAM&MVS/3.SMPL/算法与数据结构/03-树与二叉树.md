## 第一次打印 · AVL平衡搜索树

/**AVL平衡搜索树**/
typedef struct BSTNode
{
	int data;
	int bf;
	struct BSTNode *lchild,*rchild;
}BSTNode,*BSTree;
void R_Rotate(BSTree *p)
{
	BSTree lc;
	lc=(*p)->lchild;
	(*p)->lchild=lc->rchild;
	lc->rchild=*p;
}
void L_Rotate(BSTree *p)
{
	BSTree rc;
	rc=(*p)->rchild;
	(*p)->rchild=rc->lchild;
	rc->lchild=*p;
}
#define LH +1;//左高
#define EH  0;//平衡
#define RH -1;//右高
void LeftBalance(BSTree *T);
void RightBalance(BSTree *T);
/*若在平衡二叉树T不存在和e相同的关键字节点，则插入一个数据为e的																					
 *新节点，并返回1否则返回0，若因插入而使二叉排序树失去平衡，则做
 *平衡旋转处理，taller反映T长高与否
 */										  										 										 
int InsertAVL(BSTree *T,int e,int *taller)
{
	if(!(*T))
	{
		*T=(BSTree)malloc(sizeof(BSTNode));
		(*T)->data=e;
		(*T)->lchild=null;
		(*T)->rchild=null;
		(*T)->bf=EH;*taller=1;
	}
	else
	{
		if(e==(*T)->data)
		{
			*taller=0;return 0;
		}
		if(e<(*T)->data)
		{
			if(!InsertAVL(&((*T)->lchild),e,taller)) return 0;
			if(*taller)//做平衡处理
				switch((*T)->bf){
					 case LH:
						 LeftBalance(T);*taller=0;break;
					 case EH:
						 (*T)->bf=LH;*taller=1;break;
					 case RH:
						  (*T)->bf=EH;*taller=0;break;
							}
		}
	  else
	  {
		  if(!InsertAVL(&((*T)->rchild),e,taller))return 0;
		  if(*taller)
			  switch((*T)->bf){
					case LH:
						(*T)->bf=EH;*taller=0;break;
					case EH:
						(*T)->bf=RH;*taller=1;break;
					case RH:
						RightBalance(T);*taller=0;break;
							}
	  }

	}
return 1;
}
void LeftBalance(BSTree *T)
{
	BSTree	lc=(*T)->lchild;
	BSTree	rd;
		switch(lc->bf)
		{
			case LH:
				(*T)->bf=lc->bf=EH;
				R_Rotate(T);break;
			case RH://新插入的节点在*T的左孩子的右子树上,要做双旋处理
				rd=lc->rchild;
				switch(rd->bf)
				{
				case LH: (*T)->bf=RH;lc->bf=EH;break;
				case EH: (*T)->bf=lc->bf=EH; break;//疑问这里有必要这一句吗？
				case RH: (*T)->bf=EH;lc->bf=LH;break;
				}
			    rd->bf=EH;
				L_Rotate(&((*T)->lchild));
				R_Rotate(T);//这里省掉一个break;
		}
}
void RightBalance(BSTree *T)
{
	BSTree	rc=(*T)->rchild;
	BSTree	ld;
		switch(rc->bf)
		{
			case RH:
				(*T)->bf=rc->bf=EH;
				L_Rotate(T);break;
			case LH:
				ld=rc->lchild;
				switch(ld->bf)
				{
				case LH: (*T)->bf=EH;rc->bf=RH;break;
				case EH: (*T)->bf=rc->bf=EH; break;
				case RH: (*T)->bf=LH;rc->bf=EH;break;
				}
			    ld->bf=EH;
				R_Rotate(&((*T)->rchild));
				L_Rotate(T);
		}
}

	unsigned int weight;
	unsigned int parent,lchild,rchild;

void Select(HuffmanTree HT,int n,int *s1,int *s2)
{
  unsigned int m1,m2;
  int n1,n2;
  int i;
  i=1;
  while((HT[i].parent!=0)&&(i<n))i++;
  if(i<n){m1=HT[i].weight;n1=i;}
  i=i+1;
  while((HT[i].parent!=0)&&(i<=n))i++;
  i=1;
  for(i=1;i<=n;i++)
  {
	  if((HT[i].weight<m1)&&(HT[i].parent==0)){m2=m1;m1=HT[i].weight;n2=n1;n1=i;}
	  else if((HT[i].weight<m2)&&(HT[i].parent==0)
		  &&!((HT[i].weight==m1)&&(i==n1))){m2=HT[i].weight;n2=i;}
  }
  *s1=n1;
  *s2=n2;
}

     HT[s1].parent=i;HT[s2].parent=i;

int SearchBST(BiTree T,int key ,BiTree f,BiNode **pp)
{
	
	if(!T){*pp=f;return 0;}
	else if(T->data==key){*pp=T;return 1;}
	else if(T->data>key){return SearchBST(T->lchild,key,T,pp);}
	else if(T->data<key)return SearchBST(T->rchild,key,T,pp);
}
int InsertBST(BiTree *T,int key)
{
	BiNode *p;
	BiNode *s;
	if(!SearchBST(*T,key,NULL,&p))
	{	
		s=(BiTree)malloc(sizeof(BiNode));
		s->data=key;s->lchild=s->rchild=NULL;
		if(!p)*T=s;//当刚开始树为空树时，s插入为新的树根节点
		else if(key<p->data)p->lchild=s;
		else if(key>p->data)p->rchild=s;
		
	}
}
int Delete(BiNode **pp)
{
	BiNode *q;
	BiNode *r,*s;
	if(!(*pp)->rchild)
	{
		q=*pp;*pp=(*pp)->lchild;free(q);
	}
	else if(!(*pp)->lchild)
		{
		q=*pp;*pp=(*pp)->rchild;free(q);
		}
	else
	{
		q=*pp;s=(*pp)->lchild;
		while(s->rchild){q=s;s=s->rchild;}
		(*pp)->data=s->data;
		if(q!=*pp)q->rchild=s->lchild;
		else q->lchild=s->lchild;
	    Delete(&s);//直接用free(s);也是可以的
	}
}

		if((*T)->data>key)DeleteBST(&((*T)->lchild),key);
	}
}

	{
		InOrder(T->lchild);

	int i;

	int j;

	{
		if(abs(dw-sw[i-1]-sw[i])<min)

typedef struct BSTNode
{
	int data;
	int bf;
	struct BSTNode *lchild,*rchild;
}BSTNode,*BSTree;
void R_Rotate(BSTree *p)
{
	BSTree lc;
	lc=(*p)->lchild;
	(*p)->lchild=lc->rchild;
	lc->rchild=*p;
}
void L_Rotate(BSTree *p)
{
	BSTree rc;
	rc=(*p)->rchild;
	(*p)->rchild=rc->lchild;
	rc->lchild=*p;
}
#define LH +1;//左高
#define EH  0;//平衡
#define RH -1;//右高
void LeftBalance(BSTree *T);
void RightBalance(BSTree *T);
/*若在平衡二叉树T不存在和e相同的关键字节点，则插入一个数据为e的																					
 *新节点，并返回1否则返回0，若因插入而使二叉排序树失去平衡，则做
 *平衡旋转处理，taller反映T长高与否
 */										  										 										 
int InsertAVL(BSTree *T,int e,int *taller)
{
	if(!(*T))
	{
		*T=(BSTree)malloc(sizeof(BSTNode));
		(*T)->data=e;
		(*T)->lchild=null;
		(*T)->rchild=null;
		(*T)->bf=EH;*taller=1;
	}
	else
	{
		if(e==(*T)->data)
		{
			*taller=0;return 0;
		}
		if(e<(*T)->data)
		{
			if(!InsertAVL(&((*T)->lchild),e,taller)) return 0;
			if(*taller)//做平衡处理
				switch((*T)->bf){
					 case LH:
						 LeftBalance(T);*taller=0;break;
					 case EH:
						 (*T)->bf=LH;*taller=1;break;
					 case RH:
						  (*T)->bf=EH;*taller=0;break;
							}
		}
	  else
	  {
		  if(!InsertAVL(&((*T)->rchild),e,taller))return 0;
		  if(*taller)
			  switch((*T)->bf){
					case LH:
						(*T)->bf=EH;*taller=0;break;
					case EH:
						(*T)->bf=RH;*taller=1;break;
					case RH:
						RightBalance(T);*taller=0;break;
							}
	  }

	}
return 1;
}
void LeftBalance(BSTree *T)
{
	BSTree	lc=(*T)->lchild;
	BSTree	rd;
		switch(lc->bf)
		{
			case LH:
				(*T)->bf=lc->bf=EH;
				R_Rotate(T);break;
			case RH://新插入的节点在*T的左孩子的右子树上,要做双旋处理
				rd=lc->rchild;
				switch(rd->bf)
				{
				case LH: (*T)->bf=RH;lc->bf=EH;break;
				case EH: (*T)->bf=lc->bf=EH; break;//疑问这里有必要这一句吗？
				case RH: (*T)->bf=EH;lc->bf=LH;break;
				}
			    rd->bf=EH;
				L_Rotate(&((*T)->lchild));
				R_Rotate(T);//这里省掉一个break;
		}
}
void RightBalance(BSTree *T)
{
	BSTree	rc=(*T)->rchild;
	BSTree	ld;
		switch(rc->bf)
		{
			case RH:
				(*T)->bf=rc->bf=EH;
				L_Rotate(T);break;
			case LH:
				ld=rc->lchild;
				switch(ld->bf)
				{
				case LH: (*T)->bf=EH;rc->bf=RH;break;
				case EH: (*T)->bf=rc->bf=EH; break;
				case RH: (*T)->bf=LH;rc->bf=EH;break;
				}
			    ld->bf=EH;
				R_Rotate(&((*T)->rchild));
				L_Rotate(T);
		}
