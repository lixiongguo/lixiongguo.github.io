#include<stdio.h>
void CountSort(int a[],int n)//注意要求排列的序列中关键字各不相同
{
	int i,j,k;
	int c[10];
	int t1,t2;
	int r;
	for(i=0;i<10;i++)
	{
		c[i]=0;
		for(j=0;j<10;j++)
			if(a[i]>a[j])c[i]++;
	}
    k=0;
	while(k<10)
	{
		if(c[k]!=k)
		{	
			i=k;
			r=i;
			j=c[i];
			t1=a[i];
			while(j!=r)
			{
			  t2=a[j];
		      a[j]=t1;
		      i=j;
		      j=c[j];
			  c[j]=a[j];
			  t1=t2;
			}
			a[j]=t1;
			c[j]=a[j];
		}
		k++;
	}
}
main()
{
	int i;

void HeapAdjust(int s,int m)//假设a[s+1..m]都已经调整好，现在调整a[s]
{
	int j;
	int rc;
	int t;
	rc=a[s];
	for(j=2*s;j<=m;j*=2){
		if(j<m&&a[j]<a[j+1])j++;
		if(rc>=a[j])break;//找到rc应该插入的位置
		a[s]=a[j];
		a[j]=rc;//这个交换实则没必要
		s=j;
	}
	a[s]=rc;
}
void HeapSort()
{
	int i;
	int t;
	for(i=n/2;i>0;i--)//只需从最后一个非叶子节点开始，叶子节点无需考虑
		HeapAdjust(i,n);	
	for(i=n;i>1;i--)
	{
		t=a[1];
		a[1]=a[i];
		a[i]=t;
		HeapAdjust(1,i-1);
	}
}
main()
{
	int i;
	HeapSort();
for(i=1;i<=n;i++)

#include<stdio.h>
void mergeSort(int is1[],int is2[],int a,int b)// 下标，例如数组int is[5]，全部排序的调用为mergeSort(0,4)
{
	int high,mid,low;
	high=b;low=a;
if(a<b)
{
mid=(a+b)/2;
mergeSort(is1,is2,a,mid);
mergeSort(is1,is2,mid+1,b);

{
int i=low,j=mid+1,k=low;
while(i<=mid&&j<=high)
if(is1[i]<=is1[j]) // 此处为排序顺序的关键，用小于表示从小到大排序
is2[k++]=is1[i++];
else
is2[k++]=is1[j++];
while(i<=mid)
is2[k++]=is1[i++];
while(j<=high)
is2[k++]=is1[j++];
for(i=low;i<=high;i++)// 写回原数组
is1[i]=is2[i];
}

}
}

{
	int i;

	             ,0,2,3,6,8};
	int is2[10];//构造这个辅助数组是关键

{
	int i;
	int j;
	for(i=2;i<=n;i++)//假设a[1..i-1]都已经排好序,现将a[i]插入
	{
		if(a[i]<a[i-1])//要将a[i]插入
		{
			a[0]=a[i];//暂存在a[0]处
			a[i]=a[i-1];
			for(j=i-2;a[j]>a[0];j--)//移动后面的元素
				a[j+1]=a[j];
			a[j+1]=a[0];
		}
	}
}
void BInsertSort()
{
	int low,mid,high;
	int i,j;
	for(i=2;i<=n;i++)
	{
		a[0]=a[i];
		low=1;high=i-1;//在a[1..i-1]中查找a[i]的位置这里用二分查找，直接插排用的是顺序查找
		while(low<=high)
		{
			mid=(low+high)/2;
			if(a[mid]<a[0])low=mid+1;
			else high=mid-1;
		}
		for(j=i-1;j>=high+1;j--)

int fib[6];
int a[5]={0,3,6,9,10};
void GetFibonnace()
{
	fib[0]=0;fib[1]=1;
	fib[2]=1;fib[3]=2;
	fib[4]=3;fib[5]=5;
}
int FibSearch(int start ,int u,int *pos,int key)
{
	int mid;
	if(u<=1)return 0;
	mid=start+fib[u-1];
	if(a[mid]==key){*pos=mid;return 1;}
	else
	{
		if(a[mid]>key)
		{
			if(FibSearch(start,u-1,pos,key))return 1;				
			else return 0;
		}
		else
		{
			if(FibSearch(start+fib[u-1],u-2,pos,key))return 1;				
			else return 0;
		}
	}
}

/**对二元组进行排序，典型 “先快排后直插”的应用**/
#include<stdio.h>
typedef struct{int fst,snd;}node;
int Partition(node R[],int l,int h)
{
	int i=1;
	int j=h;
	R[0]=R[i];x=R[i].fst;
	while(i<j)
	{
		while(i<j&&R[j].fst>=x)j--;
		if(i<j)R[i++]=R[j];
		while(i<j&&R[i].fst<x)i++;
		if(i<j)R[j--]=R[i];
	}
	R[i]=R[0];
	return i;
}
void QuickSort(node R[],int S,T);
{
	int k;
	if(S<T)
	{
		k=Partition(R,S,T);
		QuickSort(R,S,k);
		QuickSort(R,k+1,T);
	}
}
void onepass(node R[],int i)
{
	R[0]=R[i];
	x=R[i].fst;y=R[i].snd;
	j=i-1;
	while(R[j].fst>x||R[j].fst==x&&R[j].snd>y)R[j+1]=R[j--];
	R[j+1]=R[0];
}
void InsertSort(node R[],int n)
{
	int i;
	for(i=2;i<n;i++)onepass(R[],n);
}
void Sort(node R[],int n)
{
	QuickSort(R,1,n);
	InsertSort(R,1,n);
}

while(low<high)
{
	while(low<high&&a[high]>=pivotkey)--high;
	    t=a[high];
		a[high]=a[low];

		a[low]=a[high];

a[low]=temp;

void QuickSort(int a[],int n)
{
	stack s[MAX_N];
	int top=0;
	stack p;
	int mid;
	s[++top].start=0;

			mid=Partion(a,p.start,p.end);
