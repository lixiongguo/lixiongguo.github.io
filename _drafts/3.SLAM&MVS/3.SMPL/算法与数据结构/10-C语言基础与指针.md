for(i=0;i<2*n-1;i++)
for(j=0;j<2*n-1;j++)
	if((i+j>=n-1)&&(i+j<=(n-1)+(2*n-1)-1)
   &&(j>=i-(n-1))&&(j<=i+(n-1)))
   a[i][j]=1;
else a[i][j]=0;

	for(j=0;j<2*n-1;j++)

	n=0;
	for(m[3]=379;m[3]<=987;m[3]++)

			{
				x=y%10;
				f[x]++;y=y/10;//分离各个数位。利用f[]来统计每个数的出现次数。
			}
		}
		for(t=0,k=1;k<=9;k++)//在for语句中初始化。
			if(f[k]!=1)t=1;//鉴别是否是完美式。
		if(t==0)
		{	printf("%5d+%3d=%3d",m[1],m[2],m[3]);//注意保持美观要控制输出格式。
			n++;
