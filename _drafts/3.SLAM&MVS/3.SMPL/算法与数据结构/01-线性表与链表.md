#include<stdio.h>//十七个猴子报数，报到“3”出圈，最后留下的猴子，即为猴子大王。
void game(int num[],int n)
{
	int i;
	int out;
	int count;
	count=0;
	i=0;
	out=0;
	while(out<n-1)
	{
		count+=num[i];		
		if(count==3)
		{
			out++;
			num[i]=0;
			count=0;
		}
        i++;
		if(i==n)
		i=0;
	}
}
void monkeymaking1()
{
	int num[17];
	int n=17;
	int i;
	for(i=0;i<n;i++)
		num[i]=1;
	game(num,n);
	for(i=0;i<n;i++)
		if(num[i]!=0)
			printf("No.%d\n",i+1);
}
void monkeymaking2()//数组元素用来存储下一个在圈内猴子的下表
{
	int i,test,p[17],n=17,last;//这里last即相当与链表previous节点
	for(i=0;i<n-1;i++)
		p[i]=i+1;
	p[n-1]=0;
	test=0;
	while(test!=p[test])
	{
		for(i=1;i<3;i++)
		{
			last=test;
			test=p[test];
		}//直接找出下一个要出圈的猴子

int gcd(int a,int b)
{
	int t;
	if(a<b){t=a;a=b;b=t;}
	if(a%b==0)return b;
	else return gcd(b,a%b);
}
void fun(int a[],int n,int k)
{
	int i,j,p;
	int temp;
	for(p=0;p<gcd(n,k);p++)
	{
		temp=a[p];
		i=p;
		j=(i-k+n)%n;
		while(j!=p)/**循环移位**/
		{
			a[i]=a[j];
			i=j;
			j=(j-k+n)%n;
		}
		a[i]=temp;
	}
}
