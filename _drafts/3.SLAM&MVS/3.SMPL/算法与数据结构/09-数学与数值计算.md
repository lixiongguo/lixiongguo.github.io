/**循环移位**/
#include<stdio.h>
int gcd(int a,int b)
{
	int t;
	if(a<b){t=a;a=b;b=t;}
	if(a%b==0)return b;
	else return gcd(b,a%b);
}
void fun(int a[],int n,int k)
{
	int i,j,p;
	int temp;
	for(p=0;p<gcd(n,k);p++)
	{
		temp=a[p];
		i=p;
		j=(i-k+n)%n;
		while(j!=p)/**循环移位**/
		{
			a[i]=a[j];
			i=j;
			j=(j-k+n)%n;
		}
		a[i]=temp;
	}
}
main()
{
	int a[6]={1,2,3,4,5,6};
	int n=6,k=4;
	int i;
	fun(a,6,4);
	for(i=0;i<6;i++)printf("%d\n",a[i]);

}

int max(int a,int b,int c)
{
	if(a>b&&a>c)return a;
	if(b>c)return b;
	return c;
}
int lcm(int a,int b,int c)
{
	int x,t;
	int flag;
	int i;
	x=max(a,b,c);
	t=1;
	for(i=2;i<=x;i++)
	{
		flag=1;
		while(flag==1)
		{
			flag=0;
			if(a%i==0)
			{
				a=a/i;
				flag=1;
			}
			if(b%i==0)
			{
				b=b/i;
				flag=1;
			}
			if(c%i==0)
			{
				c=c/i;
				flag=1;
			}
			if(flag==1)
			{
			  t=t*i;
			}
		}
	}
	return t;

#include<stdio.h>
#include<math.h>
typedef struct TPoint
{
	double x;
	double y;
}TPoint;
typedef struct Triangle
{
	TPoint p1;
	TPoint p2;
	TPoint p3;
}Triangle;
double Area(Triangle T)
{
	double x1=T.p1.x;
	double y1=T.p1.y;
	double x2=T.p2.x;
	double y2=T.p2.y;
	double x3=T.p3.x;
	double y3=T.p3.y;
	return  fabs(((x2*y3-y2*x3)-(x1*y3-y1*x3)+(x1*y2-y1*x2))*0.5);
}
int InTriangle1(Triangle T,TPoint p)
{
  Triangle APB,BPC,CPA;

  CPA.p1=T.p3; APB.p2=p;APB.p3=T.p1;
  if(Area(T)==Area(APB)+Area(BPC)+Area(CPA)) return 1;
  else return 0;
}
