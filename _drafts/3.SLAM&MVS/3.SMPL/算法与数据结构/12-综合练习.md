#include<iostream>
#define N 101
using namespace std;
struct match{
	int woman_love;
	int man_love;
};
int main(){
	int n;
	match m[N][N];
	bool man_engaged[N];
	bool woman_engaged[N];
	int man_engaged_to[N];
	cout << "How many couples to pair" << endl;
	cin >> n;

	cout << "woman preferences" << endl;
	for(int i=1; i<=n; i++)
		{
			cout << "enter woman["<< i << "] preferences" << endl;
			for(int j=1; j<=n; j++){
				cin >>m[i][j].woman_love;
			}

	}
	cout << "man preferences" << endl;
	for(int i=1; i<=n; i++)
		{
			cout << "enter man["<< i << "] preferences" << endl;
			for(int j=1; j<=n; j++){
				cin >> m[i][j].man_love;
			}
	}
	for(int i=1; i<=n; i++){
		woman_engaged[i] = false;
		man_engaged[i] = false;
		man_engaged_to[i] = 0;
	}
	bool over_flag = false;
	int k=1;
	while(1){
		over_flag  = true;
		for(k=1;k<=n;k++)
		{
			if(!woman_engaged[k])
			{
				over_flag = false;
				break;
			}
		}
		if(over_flag)break;
		int s=1;
		while(s <= n)
		{
			int i=1;
			for(; i<=n; i++)
			if(m[k][i].woman_love == s)break;

			if(!man_engaged[i]){
				man_engaged[i] =true;
				man_engaged_to[i] = k;
				woman_engaged[k] = true;
				break;
			}
			else{
				int l = man_engaged_to[i];
				if(m[l][i].man_love > m[k][i].man_love)
					{
						man_engaged_to[i] = k;
						woman_engaged[k] = true;
						man_engaged[k] = true;
						break;
				}
				else s++;
			}
		}
		
	}
	
}

//给出前序序列与中序序列，求指定两节点的共同祖先
#include<stdio.h>
char getAncestor()
{
	char str1[100]="ABDEHCFIJGK";
	char str2[100]="DBHEAIFJCKG";
	int len_1=11,len_2=11;
	char x='I',y='G';
	char c;
	int t;
	int i;
	int count;
	int low1,high1;
	int low2,high2;
	low1=0;high1=len_1-1;
	low2=0;high2=len_2-1;
	count=0;
	t=0;
	while(1)
	{
		c=str1[low1];
		if(x==c)return c;
		if(y==c)return c;
		for(i=low2;i<=high2;i++)
		{
			if(str2[i]!=c)count++;
			else break;
			if(str2[i]==x||str2[i]==y)t++;
		}
		if(t==2){low2=low2;high2=count-1;low1=low1+1;high1=low1+(high2-low2);}
		else if(t==1){return c;}
		else if(t==0){low2=count+1;high2=high2;high1=high1;low1=high1-(high2-low2);}
		t=0;
	}
}
main()
{
	char c;
