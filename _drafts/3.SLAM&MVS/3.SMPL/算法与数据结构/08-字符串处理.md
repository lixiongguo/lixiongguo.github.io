/**字符串Replace的实现**/
#include<stdio.h>
#include<string.h>
void replace(char *s,char *t,char *v)
{
	int i,j,k;
	int flag;
	int len_t=strlen(t),len_v=strlen(v);
	i=0;
	while(i<strlen(s))
	{
		flag=1;
		for(j=0;j<len_t;j++)
		{
			if(s[i+j]!=t[j])
			{
				flag=0;break;
			}
		}
		if(flag==1)
		{
			if(len_v>=len_t)
			{
				for(j=strlen(s);j>=i+len_t;j--)//第strlen(s)个元素是'\n';
					s[j+len_v-len_t]=s[j];				
				for(j=0;j<len_v;j++)
					s[i+j]=v[j];
				i=i+len_v;
			}			
		 else
			{
				for(j=i+len_t;j<=strlen(s);j++)
					s[j-(len_t-len_v)]=s[j];
				for(j=0;j<len_v;j++)
					s[i+j]=v[j];
				i=i+len_v;
			}
			
		}
		else i=i+1;
	}
}
main()
{
	char s[80]="abcabcabc";
	char t[10]="abc";
	char v[10]="de";
	replace(s,t,v);
	puts(s);
}
